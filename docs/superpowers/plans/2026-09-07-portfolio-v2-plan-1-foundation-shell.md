# Portfolio v2 — Plan 1: Foundation & Shell Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A deployed, classic-Macintosh-styled single-page site whose one open window is Samuel's résumé, rendered from `content/resume.md` as real static HTML, working with JavaScript disabled, with a working desktop shell (menu bar, boot, sound, draggable window) when JS is on.

**Architecture:** One Vite + React + TypeScript app, one route (`/`), statically generated by `vite-react-ssg` so every window's content ships as semantic HTML in `dist/index.html`. React hydrates and progressively enhances that HTML into a draggable classic-Mac desktop. With JS off (or on mobile) the same markup reads as a plain vertical stack of documents. A build step parses `content/*.md` into typed data the React components consume. A GitHub Action builds and deploys `dist/` to GitHub Pages.

**Tech Stack:** Vite 5, React 18, TypeScript 5, `vite-react-ssg` (static generation), Vitest + @testing-library/react (unit), Playwright + @axe-core/playwright (e2e / visual / a11y), `gray-matter` + `marked` (markdown), Web Audio API (chime), plain CSS with CSS Modules (no CSS framework, no animation library — zoom-rects and dissolves are hand-rolled with the Web Animations API).

**Spec:** `docs/superpowers/specs/2026-09-07-portfolio-v2-classic-mac-design.md`

## Global Constraints

Every task's requirements implicitly include these. Values copied verbatim from the spec.

- **Strictly 1-bit.** Only `#000` and `#fff` as colour values. Greys are achieved by ordered dithering (CSS background patterns), never a grey value. No accent colour anywhere. Interaction feedback (hover / active / selected / focus) is a full invert of the element's fill and text.
- **Real static HTML.** Every window's content ships as real semantic HTML in the served markup. The desktop shell hydrates over pre-rendered content; it never generates it. A recruiter with JS disabled, an ATS scraper, or a screen reader must get the full résumé as an ordinary document.
- **Vite `base: '/'`** — the repo is a GitHub user page served at the domain root.
- **`content/resume.md` is the single résumé source of truth.** Parsed at build; the Résumé window renders it; a future Terminal `cat resume.md` will print it raw; the PDF derives from it.
- **Fonts:** authentic Macintosh families via free webfont clones — Chicago (menus, window titles, buttons), Geneva (UI labels, résumé fields), New York (long-form reading prose), Monaco (monospace). Self-hosted, no external font CDN.
- **`prefers-reduced-motion: reduce`** turns every animation (zoom-rects, dissolves, boot build-in, flicker) into an instant state change; drag becomes live-position instead of outline.
- **Deploy:** GitHub Action runs `npm ci && npm run build` and publishes `dist/` to GitHub Pages. `Resume.pdf` stays reachable at the served root. `public/.nojekyll` present.
- **Copy style:** plain, first person, dry, specific. No buzzword soup, no LinkedIn headline-ese. Sparing with em dashes (taste, not a hard ban).
- **Content accuracy (from the audit):** Apple dates are "Sept 2023 – Present" (never "2023 – 2026"); TMU is "2019 – 2024" everywhere; Kotlin appears in skills; Solr is categorised as search, not AI; do not write "fine-tuned" before RoBERTa; the "2B+ records per hour" figure is included.

```yaml
name: Deploy site
on:
  push:
    branches: [main, new-and-improved]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
      - run: npm ci
      - run: npm run build
      - run: npm run prerender:assert
      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist

  deploy:
    needs: build
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
```

```ts
// src/shell/windows.ts
export type WindowId = 'resume' | 'readme' | 'terminal' | 'projects' | 'puzzle' | 'about'

export interface WindowState {
  id: WindowId; open: boolean; z: number
  x: number; y: number; w: number; h: number
  shaded: boolean; minZoom: { x: number; y: number } | null
}
export interface DesktopState { windows: Record<WindowId, WindowState>; nextZ: number }

const DEFAULTS: Record<WindowId, Pick<WindowState, 'x' | 'y' | 'w' | 'h'>> = {
  resume:   { x: 80,  y: 44,  w: 560, h: 620 },
  readme:   { x: 120, y: 24,  w: 420, h: 260 },
  terminal: { x: 220, y: 120, w: 620, h: 380 },
  projects: { x: 160, y: 60,  w: 700, h: 520 },
  puzzle:   { x: 300, y: 140, w: 240, h: 280 },
  about:    { x: 260, y: 160, w: 360, h: 220 },
}

function make(id: WindowId): WindowState {
  return { id, open: false, z: 0, shaded: false, minZoom: null, ...DEFAULTS[id] }
}

export const initialDesktop: DesktopState = {
  windows: {
    resume: make('resume'), readme: make('readme'), terminal: make('terminal'),
    projects: make('projects'), puzzle: make('puzzle'), about: make('about'),
  },
  nextZ: 1,
}

export type WindowAction =
  | { type: 'open'; id: WindowId; from?: { x: number; y: number } }
  | { type: 'close'; id: WindowId }
  | { type: 'focus'; id: WindowId }
  | { type: 'move'; id: WindowId; x: number; y: number }
  | { type: 'resize'; id: WindowId; w: number; h: number }
  | { type: 'toggleShade'; id: WindowId }

export function windowsReducer(state: DesktopState, action: WindowAction): DesktopState {
  const w = state.windows[action.id]
  switch (action.type) {
    case 'open':
      return {
        ...state, nextZ: state.nextZ + 1,
        windows: { ...state.windows, [action.id]: { ...w, open: true, z: state.nextZ, minZoom: action.from ?? null } },
      }
    case 'close':
      return { ...state, windows: { ...state.windows, [action.id]: { ...w, open: false } } }
    case 'focus':
      if (!w.open) return state
      return {
        ...state, nextZ: state.nextZ + 1,
        windows: { ...state.windows, [action.id]: { ...w, z: state.nextZ } },
      }
    case 'move':
      return { ...state, windows: { ...state.windows, [action.id]: { ...w, x: action.x, y: action.y } } }
    case 'resize':
      return { ...state, windows: { ...state.windows, [action.id]: { ...w, w: Math.max(240, action.w), h: Math.max(120, action.h) } } }
    case 'toggleShade':
      return { ...state, windows: { ...state.windows, [action.id]: { ...w, shaded: !w.shaded } } }
  }
}

export function frontWindow(state: DesktopState): WindowId | null {
  const open = Object.values(state.windows).filter((w) => w.open)
  if (!open.length) return null
  return open.reduce((a, b) => (b.z > a.z ? b : a)).id
}
```

```ts
// src/shell/useDrag.ts
import { useRef, useCallback } from 'react'

export function useDrag(onDelta: (dx: number, dy: number) => void) {
  const origin = useRef<{ x: number; y: number } | null>(null)

  const onPointerMove = useCallback((e: PointerEvent) => {
    if (!origin.current) return
    onDelta(e.clientX - origin.current.x, e.clientY - origin.current.y)
    origin.current = { x: e.clientX, y: e.clientY }
  }, [onDelta])

  const onPointerUp = useCallback(() => {
    origin.current = null
    window.removeEventListener('pointermove', onPointerMove)
    window.removeEventListener('pointerup', onPointerUp)
  }, [onPointerMove])

  const onPointerDown = useCallback((e: React.PointerEvent) => {
    if (e.button !== 0) return
    origin.current = { x: e.clientX, y: e.clientY }
    window.addEventListener('pointermove', onPointerMove)
    window.addEventListener('pointerup', onPointerUp)
  }, [onPointerMove, onPointerUp])

  return { onPointerDown }
}
```

```tsx
// src/shell/Window.tsx
import type { WindowState } from './windows'
import { useDrag } from './useDrag'
import styles from './Window.module.css'

interface WindowProps {
  state: WindowState
  title: string
  onFocus(): void
  onClose(): void
  onMove(x: number, y: number): void
  onResize(w: number, h: number): void
  onToggleShade(): void
  children: React.ReactNode
}

export default function Window(p: WindowProps) {
  const { state } = p
  const drag = useDrag((dx, dy) => p.onMove(state.x + dx, state.y + dy))
  const grow = useDrag((dx, dy) => p.onResize(state.w + dx, state.h + dy))

  return (
    <section
      role="dialog"
      aria-label={p.title}
      className={styles.window}
      style={{ left: state.x, top: state.y, width: state.w, height: state.shaded ? undefined : state.h, zIndex: state.z }}
      onPointerDown={p.onFocus}
    >
      <div
        className={styles.titlebar}
        data-testid="titlebar"
        onPointerDown={drag.onPointerDown}
```

```ts
export function zoomFrames(from: DOMRectInit, to: DOMRectInit): Keyframe[] {
  return [
    { transform: `translate(${from.x}px, ${from.y}px) scale(${from.width || 1}, ${from.height || 1})`, opacity: 1 },
```

```tsx
// src/shell/ZoomRect.tsx
import { useEffect, useRef } from 'react'
import { zoomFrames } from './zoomRect'

const reduce = () =>
  typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches

export default function ZoomRect(props: {
  from: { x: number; y: number }
  to: { x: number; y: number; w: number; h: number }
  onDone(): void
}) {
  const ref = useRef<HTMLDivElement>(null)
  useEffect(() => {
    if (reduce() || !ref.current) { const id = requestAnimationFrame(props.onDone); return () => cancelAnimationFrame(id) }
    const anim = ref.current.animate(
      zoomFrames(
        { x: props.from.x, y: props.from.y, width: 0, height: 0 },
        { x: props.to.x, y: props.to.y, width: props.to.w, height: props.to.h },
      ),
      { duration: 180, easing: 'steps(6, end)' },
    )
    anim.onfinish = props.onDone
    return () => anim.cancel()
  }, [props])
  return (
    <div
      ref={ref}
      aria-hidden
      style={{
        position: 'fixed', left: 0, top: 0, width: 1, height: 1,
        border: '1px solid #000', transformOrigin: 'top left', pointerEvents: 'none', zIndex: 9999,
      }}
    />
  )
}
```

```ts
// src/shell/useBoot.ts
import { useCallback, useEffect, useState } from 'react'

const reduce = () =>
  typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches

export function useBoot() {
  const done = () =>
    typeof sessionStorage !== 'undefined' && sessionStorage.getItem('booted') === '1'
  const [booting, setBooting] = useState(() => !done() && !reduce())

  const finish = useCallback(() => {
    try { sessionStorage.setItem('booted', '1') } catch { /* ignore */ }
    setBooting(false)
  }, [])

  useEffect(() => {
    if (!booting) return
    const id = setTimeout(finish, 1500)
    return () => clearTimeout(id)
  }, [booting, finish])

  return { booting, skip: finish }
}
```

```ts
// src/shell/sound.ts
const A = 440
const noteHz = (semitonesFromA4: number) => A * Math.pow(2, semitonesFromA4 / 12)
// C4 E4 G4 C5 relative to A4: -9, -5, -2, +3
const CHORD = [-9, -5, -2, 3].map(noteHz)

export class ChimeEngine {
  private ctxFactory: () => AudioContext
  private ctx: AudioContext | null = null
  private muted = false

  constructor(ctxFactory?: () => AudioContext) {
    this.ctxFactory =
      ctxFactory ?? (() => new (window.AudioContext || (window as any).webkitAudioContext)())
  }
  setMuted(m: boolean) { this.muted = m }

  private ac(): AudioContext {
    if (!this.ctx) this.ctx = this.ctxFactory()
    return this.ctx
  }

  playChime() {
    if (this.muted) return
    const ac = this.ac()
    const t0 = ac.currentTime
    CHORD.forEach((hz, i) => {
      const osc = ac.createOscillator()
      const gain = ac.createGain()
      osc.type = 'triangle'
      osc.frequency.setValueAtTime(hz, t0 + i * 0.06)
      gain.gain.setValueAtTime(0.0001, t0 + i * 0.06)
      gain.gain.exponentialRampToValueAtTime(0.18, t0 + i * 0.06 + 0.02)
      gain.gain.exponentialRampToValueAtTime(0.0001, t0 + i * 0.06 + 1.6)
      osc.connect(gain); gain.connect(ac.destination)
      osc.start(t0 + i * 0.06); osc.stop(t0 + i * 0.06 + 1.7)
    })
  }

  beep() {
    if (this.muted) return
    const ac = this.ac()
    const t0 = ac.currentTime
```

```tsx
// src/shell/useSound.tsx
import { createContext, useContext, useMemo, useState, useCallback } from 'react'
import { ChimeEngine } from './sound'

interface SoundApi { muted: boolean; toggle(): void; chime(): void; beep(): void }
const Ctx = createContext<SoundApi | null>(null)

export function SoundProvider({ children }: { children: React.ReactNode }) {
  const engine = useMemo(() => new ChimeEngine(), [])
  const [muted, setMuted] = useState(() => {
    try { return localStorage.getItem('muted') === '1' } catch { return false }
  })
  engine.setMuted(muted)

  const toggle = useCallback(() => {
    setMuted((m) => {
      const next = !m
      try { localStorage.setItem('muted', next ? '1' : '0') } catch { /* ignore */ }
      engine.setMuted(next)
      return next
    })
  }, [engine])

  const api = useMemo<SoundApi>(
    () => ({ muted, toggle, chime: () => engine.playChime(), beep: () => engine.beep() }),
    [muted, toggle, engine],
  )
  return <Ctx.Provider value={api}>{children}</Ctx.Provider>
}

export function useSound(): SoundApi {
  const v = useContext(Ctx)
  if (!v) throw new Error('useSound must be used within SoundProvider')
  return v
}
```
